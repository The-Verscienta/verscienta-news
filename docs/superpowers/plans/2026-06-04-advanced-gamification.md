# Advanced Gamification Implementation Plan

> **Historical — standalone Verscienta app.** This file describes `Verscienta.*` LiveView work. It is not the Kiln overlay and it is not Slice 1. Current scope is [`docs/slice-1.md`](../../slice-1.md).


> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Layer the full advanced-gamification loop — quests, streaks, leaderboards, reward store, referrals — on top of the existing Phase 1 points/badges engine, all fed by a single `GamificationEvent` stream.

**Architecture:** A new `Verscienta.Gamification.Events.record/4` is the single producer of `GamificationEvent` rows. It populates user-local `local_date`, enforces daily per-source caps, enqueues the existing `AwardWorker` for points-worthy events, and enqueues a new `ProgressWorker` that advances streaks + quest assignments and emits notifications. Cron-driven Oban workers handle quest assignment, weekly streak-freeze refill, leaderboard snapshots, and hourly anomaly scans. Reward redemption goes through `Store.redeem/2`, a single DB transaction that writes a negative `PointsEntry`, applies the effect, and rate-limits. Referrals are captured via cookie at signup and qualify on first daily-quest completion.

**Tech Stack:** Elixir 1.18.4 / OTP 27.3.4, Phoenix 1.8, Ash 3.x, AshPostgres, Oban, Tailwind v4, daisyUI. Tests via ExUnit + `Oban.Testing` (`testing: :manual`) + `Phoenix.LiveViewTest`. All `mix` runs in Docker.

**Conventions (apply to every task):**
- Run all `mix` commands **in Docker**: `docker compose exec app mix <...>`. Never run mix on the host.
- **Never hand-write migrations.** Use `docker compose exec app mix ash.codegen <descriptive_name>`. Review the generated migration before committing.
- Before each commit: `mix format`; LF line endings; no `.bak` files.
- Each commit body ends with: `Co-Authored-By: Claude Opus 4.8 (1M context) <noreply@anthropic.com>`
- Test DB resets use `MIX_ENV=test mix ecto.drop -r Verscienta.Repo && mix ecto.create -r Verscienta.Repo && mix ash.migrate` (NOT `ash.setup` — seeding contaminates count assertions; see `[[verscienta-test-db-seed-contamination]]`).
- Work on branch `phase-2-advanced-gamification` (created in Task 0).
- 8 categories used by predicates and category leaderboards: `:nutrition, :herbalism, :mindfulness, :movement, :functional_medicine, :tcm, :ayurveda, :environment` (lowercased slugs in `metadata["category"]`).
- Idempotency keys (`ref`, `source_ref`, notification subject_id) are always strings like `"event:<uuid>"`, `"quest:<assignment_id>"`, etc.
- PubSub topics: `"user:<id>"` (per-user — already exists), `"leaderboards"` (new — global).

defmodule Verscienta.Gamification.GamificationEvent do
  @moduledoc "Append-only behavior event stream powering streaks, quests, and leaderboards."
  use Ash.Resource,
    otp_app: :verscienta,
    domain: Verscienta.Gamification,
    data_layer: AshPostgres.DataLayer

  @kinds [
    :article_published,
    :comment_posted,
    :reaction_received,
    :reading,
    :followed_user,
    :bookmarked,
    :reposted,
    :joined_group,
    :invited_friend_signed_up
  ]
  @subject_types [:article, :comment, :user, :group, nil]

  def kinds, do: @kinds
  def subject_types, do: @subject_types

  postgres do
    table "gamification_events"
    repo Verscienta.Repo

    custom_indexes do
      index [:user_id, :local_date]
      index [:user_id, :kind, :local_date]
      index [:occurred_at]
    end
  end

  actions do
    defaults [:read]

    create :create do
      accept [:user_id, :kind, :ref, :subject_type, :subject_id, :metadata, :occurred_at, :local_date]
    end
  end

  attributes do
    uuid_primary_key :id
    attribute :kind, :atom, allow_nil?: false, public?: true, constraints: [one_of: @kinds]
    attribute :ref, :string, allow_nil?: false, public?: true

    attribute :subject_type, :atom,
      allow_nil?: true,
      public?: true,
      constraints: [one_of: Enum.reject(@subject_types, &is_nil/1)]

    attribute :subject_id, :string, allow_nil?: true, public?: true
    attribute :metadata, :map, allow_nil?: false, default: %{}, public?: true
    attribute :occurred_at, :utc_datetime_usec, allow_nil?: false, public?: true
    attribute :local_date, :date, allow_nil?: false, public?: true
    create_timestamp :inserted_at
  end

  relationships do
    belongs_to :user, Verscienta.Accounts.User, allow_nil?: false
  end

  identities do
    identity :unique_event, [:user_id, :kind, :ref]
  end
end

  Single producer of GamificationEvent rows. Computes per-user local date,
  enforces daily per-source caps, enqueues AwardWorker for points-worthy
  events, enqueues ProgressWorker for streak/quest progression, and

  alias Verscienta.Accounts.User
  alias Verscienta.Gamification
  alias Verscienta.Gamification.GamificationAudit
  alias Verscienta.Gamification.GamificationEvent
  alias Verscienta.Gamification.Points

  @daily_caps %{
    article_published: 5,
    comment_posted: 30,
    reaction_received: 50,
    reading: 30,
    followed_user: 20,
    bookmarked: 20,
    reposted: 20,
    joined_group: 20,
    invited_friend_signed_up: 100
  }

  @doc """
  Record a behavior event. Returns one of:
    * `{:ok, :recorded}` — event was inserted
    * `{:ok, :duplicate}` — event already existed (idempotent no-op)
    * `{:ok, :capped}` — daily cap hit; event NOT inserted, audit row written
    * `{:error, reason}` — unrecoverable failure

  Options:
    * `:metadata` — map merged into the event row
    * `:subject_type`, `:subject_id` — optional polymorphic pointer
    * `:occurred_at` — defaults to `DateTime.utc_now/0`
  """
  def record(user_id, kind, ref, opts \\ []) do
    occurred_at = Keyword.get(opts, :occurred_at, DateTime.utc_now())
    metadata = Keyword.get(opts, :metadata, %{}) |> stringify_keys()
    subject_type = Keyword.get(opts, :subject_type)
    subject_id = Keyword.get(opts, :subject_id)

    with {:ok, user} <- load_user(user_id),
         local_date <- compute_local_date(user.timezone, occurred_at),
         :ok <- check_cap(user_id, kind, local_date) do
      do_insert(user_id, kind, ref, subject_type, subject_id, metadata, occurred_at, local_date)
    else
      {:capped, _} ->
        write_cap_audit(user_id, kind, occurred_at)
        {:ok, :capped}

      {:error, _} = err ->
        err
    end
  end

  defp load_user(user_id) do
    case Ash.get(User, user_id, authorize?: false) do
      {:ok, u} -> {:ok, u}
      err -> err
    end
  end

  defp compute_local_date(nil, occurred_at), do: compute_local_date("Etc/UTC", occurred_at)

  defp compute_local_date(tz, occurred_at) do
    occurred_at
    |> DateTime.shift_zone!(tz)
    |> DateTime.to_date()
  rescue
    _ -> DateTime.to_date(occurred_at)
  end

  defp cap(kind), do: Map.get(@daily_caps, kind)

  defp check_cap(user_id, kind, local_date) do
    case cap(kind) do
      nil ->
        :ok

      limit ->
        count =
          GamificationEvent
          |> Ash.Query.filter(user_id == ^user_id and kind == ^kind and local_date == ^local_date)
          |> Ash.read!(authorize?: false)
          |> length()

        if count >= limit, do: {:capped, count}, else: :ok
    end
  end

  defp do_insert(user_id, kind, ref, subject_type, subject_id, metadata, occurred_at, local_date) do
    attrs = %{
      user_id: user_id,
      kind: kind,
      ref: ref,
      subject_type: subject_type,
      subject_id: subject_id,
      metadata: metadata,
      occurred_at: occurred_at,
      local_date: local_date
    }

    case GamificationEvent
         |> Ash.Changeset.for_create(:create, attrs)
         |> Ash.create(authorize?: false) do
      {:ok, event} ->
        maybe_enqueue_award(user_id, kind, ref)
        enqueue_progress(event.id)
        Gamification.broadcast(user_id, {:gamification_event, user_id, event})
        {:ok, :recorded}

      {:error, %Ash.Error.Invalid{errors: errs}} = err ->
        if Enum.any?(errs, &match?(%{constraint: :unique_event}, &1)) or
             Enum.any?(errs, fn e -> Map.get(e, :message) =~ "has already been taken" end),
           do: {:ok, :duplicate},
           else: err

      err ->
        err
    end
  end

  defp maybe_enqueue_award(user_id, kind, ref) do
    if kind in Points.sources() do
      %{user_id: user_id, source: Atom.to_string(kind), source_ref: ref}
      |> Verscienta.Gamification.Workers.AwardWorker.new()
      |> Oban.insert()
    end
  end

  defp enqueue_progress(event_id) do
    %{event_id: event_id}
    |> Verscienta.Gamification.Workers.ProgressWorker.new()
    |> Oban.insert()
  end

  defp write_cap_audit(user_id, kind, occurred_at) do
    details = %{"kind" => Atom.to_string(kind)}

    GamificationAudit
    |> Ash.Changeset.for_create(:create, %{
      user_id: user_id,
      kind: :source_cap_hit,
      severity: :info,
      details: details,
      occurred_at: occurred_at
    })
    |> Ash.create(authorize?: false)
  end

  defp stringify_keys(map) when is_map(map) do
    Map.new(map, fn
      {k, v} when is_atom(k) -> {Atom.to_string(k), v}
      kv -> kv
    end)
  end
end

defmodule Verscienta.Gamification.Streaks do
  @moduledoc "Pure streak math + DB-bound apply via ProgressWorker."

  @milestones [7, 30, 100, 365]

  def milestones, do: @milestones

  @doc """
  Pure: given the current streak state (a map with the same keys as the
  Streak resource) and the date of the new event, return the updated state.
  """
  def extend(%{last_active_date: nil} = s, date) do
    s
    |> Map.put(:current_length, 1)
    |> Map.put(:last_active_date, date)
    |> Map.update!(:longest_length, &max(&1, 1))
  end

  def extend(%{last_active_date: same} = s, date) when same == date, do: s

  def extend(%{last_active_date: last} = s, date) do
    gap = Date.diff(date, last)

    cond do
      gap == 1 ->
        new_len = s.current_length + 1
        s
        |> Map.put(:current_length, new_len)
        |> Map.put(:last_active_date, date)
        |> Map.update!(:longest_length, &max(&1, new_len))

      gap == 2 and s.freezes_available > 0 ->
        new_len = s.current_length + 1
        s
        |> Map.put(:current_length, new_len)
        |> Map.put(:last_active_date, date)
        |> Map.put(:freezes_available, s.freezes_available - 1)
        |> Map.put(:last_freeze_used_on, Date.add(last, 1))
        |> Map.update!(:longest_length, &max(&1, new_len))

      true ->
        s
        |> Map.put(:current_length, 1)
        |> Map.put(:last_active_date, date)
        |> Map.update!(:longest_length, &max(&1, 1))
    end
  end

  @doc "If `new.current_length` newly hit a milestone, return that milestone; else nil."
  def milestone_crossed?(old, new) do
    Enum.find(@milestones, fn m -> old.current_length < m and new.current_length >= m end)
  end
end

  alias Verscienta.Gamification.Streaks

    case Ash.get(GamificationEvent, event_id, authorize?: false) do
      {:ok, event} ->
        :ok = advance_streak(event)
        # Future tasks add: advance_quests(event); maybe_qualify_referral(event)
        :ok

      {:error, _} ->
        :ok
    end
  end

    streak = load_or_init(event.user_id)
    old = to_state(streak)
    new = Streaks.extend(old, event.local_date)

    if new != old do
      apply_update!(streak, new)
      maybe_notify_milestone(event.user_id, old, new)
    end

    :ok

      {:ok, s} ->
        s

      |> Ash.update!(authorize?: false)

      accept [:user_id, :quest_id, :period_start_date, :period_end_date,
              :goal_count, :points_reward]
    end

    update :bump_progress do
      description "Atomic increment of progress; sets completed_at when goal hit."
      accept []
      require_atomic? false
      argument :now, :utc_datetime_usec, allow_nil?: false

      change fn changeset, _ ->
        data = changeset.data
        new_progress = data.progress + 1
        changeset = Ash.Changeset.force_change_attribute(changeset, :progress, new_progress)

        if new_progress >= data.goal_count and is_nil(data.completed_at) do
          now = Ash.Changeset.get_argument(changeset, :now)
          Ash.Changeset.force_change_attribute(changeset, :completed_at, now)
        else
          changeset
        end
      end
    end
  end

  attributes do
    uuid_primary_key :id
    attribute :period_start_date, :date, allow_nil?: false, public?: true
    attribute :period_end_date, :date, allow_nil?: false, public?: true
    attribute :progress, :integer, allow_nil?: false, default: 0, public?: true

    attribute :points_reward, :integer, allow_nil?: false, public?: true
    attribute :completed_at, :utc_datetime_usec, allow_nil?: true, public?: true

defmodule Verscienta.Gamification.Workers.QuestAssignerWorker do
  @moduledoc """
  Hourly cron: for every user whose local-midnight has just passed since the
  previous run, assign every active daily Quest. On the user's local Monday
  also assign every active weekly Quest. Idempotent via the unique identity
  on (user_id, quest_id, period_start_date).
  """
  use Oban.Worker, queue: :default, max_attempts: 3, unique: [period: 1800]

  require Ash.Query

  alias Verscienta.Accounts.User
  alias Verscienta.Gamification.Quest
  alias Verscienta.Gamification.QuestAssignment

  @impl Oban.Worker
  def perform(%Oban.Job{}) do
    quests = active_quests()

    User
    |> Ash.read!(authorize?: false)
    |> Enum.each(&assign_for_user(&1, quests))

    :ok
  end

  defp active_quests do
    Quest
    |> Ash.Query.filter(active? == true)
    |> Ash.read!(authorize?: false)
  end

  defp assign_for_user(user, quests) do
    today = local_today(user.timezone)
    monday? = Date.day_of_week(today) == 1

    quests
    |> Enum.each(fn q ->
      cond do
        q.cadence == :daily ->
          upsert_assignment(user.id, q, today, today)

        q.cadence == :weekly and monday? ->
          upsert_assignment(user.id, q, today, Date.add(today, 6))

        true ->
          :ok
      end
    end)
  end

  defp upsert_assignment(user_id, quest, start_date, end_date) do
    attrs = %{
      user_id: user_id,
      quest_id: quest.id,
      period_start_date: start_date,
      period_end_date: end_date,
      goal_count: quest.goal_count,
      points_reward: quest.points_reward
    }

    QuestAssignment
    |> Ash.Changeset.for_create(:create, attrs)
    |> Ash.create(authorize?: false)
  rescue
    _ -> :ok
  end

  defp local_today(tz) do
    DateTime.utc_now()
    |> DateTime.shift_zone!(tz || "Etc/UTC")
    |> DateTime.to_date()
  rescue
    _ -> Date.utc_today()
  end

  @doc "Public helper: do an immediate same-day assignment for one user (used by Events.record fallback)."
  def assign_for(user) do
    assign_for_user(user, active_quests())
    :ok
  end
end

defmodule Verscienta.Gamification.Leaderboards do
  @moduledoc "Score aggregation for leaderboard snapshots and live queries."

  import Ecto.Query

  alias Verscienta.Repo

  @top_n 100

  @doc """
  Compute the top `@top_n` entries for the given board over the given window.

  * `scope` — `:global` or `:category`
  * `category` — `nil` for `:global`, slug string for `:category`
  * `window` — `:weekly | :monthly | :all_time`
  * `now` — defaults to `DateTime.utc_now/0`
  """
  def compute(scope, category, window, now \\ DateTime.utc_now()) do
    {start_at, end_at} = window_bounds(window, now)
    do_compute(scope, category, start_at, end_at)
  end

  @doc "Returns the {period_start_date, period_end_date} for the *closed* prior period."
  def closed_period(window, now \\ DateTime.utc_now()) do
    today = DateTime.to_date(now)

    case window do
      :weekly ->
        # Most recently closed Mon-Sun
        last_sunday = Date.add(today, -Date.day_of_week(today))
        last_monday = Date.add(last_sunday, -6)
        {last_monday, last_sunday}

      :monthly ->
        first_of_this_month = %Date{today | day: 1}
        last_of_last_month = Date.add(first_of_this_month, -1)
        first_of_last_month = %Date{last_of_last_month | day: 1}
        {first_of_last_month, last_of_last_month}

      :all_time ->
        {Verscienta.Gamification.LeaderboardSnapshot.sentinel_date(),
         Verscienta.Gamification.LeaderboardSnapshot.sentinel_date()}
    end
  end

  defp window_bounds(:all_time, _now), do: {nil, nil}

  defp window_bounds(:weekly, now) do
    today = DateTime.to_date(now)
    monday = Date.beginning_of_week(today)
    {date_to_dt(monday), now}
  end

  defp window_bounds(:monthly, now) do
    today = DateTime.to_date(now)
    first = %Date{today | day: 1}
    {date_to_dt(first), now}
  end

  defp date_to_dt(date), do: DateTime.new!(date, ~T[00:00:00], "Etc/UTC")

  defp do_compute(:global, _category, start_at, end_at) do
    query =
      from p in "points_entries",
        where: p.amount > 0,
        group_by: p.user_id,
        select: %{user_id: p.user_id, score: sum(p.amount)},
        order_by: [desc: sum(p.amount)],
        limit: @top_n

    query = apply_window(query, start_at, end_at)

    Repo.all(query)
    |> Enum.with_index(1)
    |> Enum.map(fn {row, rank} -> Map.put(row, :rank, rank) end)
  end

  defp do_compute(:category, category, start_at, end_at) do
    query =
      from p in "points_entries",
        join: e in "gamification_events",
        on: e.ref == p.source_ref and e.user_id == p.user_id,
        where: p.amount > 0 and fragment("?->>'category' = ?", e.metadata, ^category),
        group_by: p.user_id,
        select: %{user_id: p.user_id, score: sum(p.amount)},
        order_by: [desc: sum(p.amount)],
        limit: @top_n

    query = apply_window(query, start_at, end_at)

    Repo.all(query)
    |> Enum.with_index(1)
    |> Enum.map(fn {row, rank} -> Map.put(row, :rank, rank) end)
  end

  defp apply_window(query, nil, _), do: query

  defp apply_window(query, start_at, end_at) do
    from p in query, where: p.inserted_at >= ^start_at and p.inserted_at <= ^end_at
  end
end

defmodule Verscienta.Gamification.Workers.LeaderboardSnapshotWorker do
  @moduledoc "Cron worker: closes weekly/monthly boards at UTC boundary; recomputes all-time daily."
  use Oban.Worker, queue: :default, max_attempts: 3, unique: [period: 3600]

  alias Verscienta.Gamification.LeaderboardSnapshot
  alias Verscienta.Gamification.Leaderboards

  @categories ["nutrition", "herbalism", "mindfulness", "movement",
               "functional_medicine", "tcm", "ayurveda", "environment"]

  @impl Oban.Worker
  def perform(%Oban.Job{args: args}) do
    now = DateTime.utc_now()
    windows = which_windows(now, args)

    Enum.each(windows, fn window ->
      snapshot(:global, nil, window, now)
      Enum.each(@categories, fn cat -> snapshot(:category, cat, window, now) end)
    end)

    :ok
  end

  defp which_windows(_now, %{"window" => w}), do: [String.to_existing_atom(w)]

  defp which_windows(now, _) do
    today = DateTime.to_date(now)
    base = [:all_time]
    base = if Date.day_of_week(today) == 1, do: base ++ [:weekly], else: base
    if today.day == 1, do: base ++ [:monthly], else: base
  end

  defp snapshot(scope, category, window, now) do
    entries = Leaderboards.compute(scope, category, window, now)

    {start_date, end_date} =
      case window do

defmodule Verscienta.Gamification.Cache do
  @moduledoc "ETS-backed 60s cache for live leaderboard queries."

  @table :verscienta_gamification_cache

  def start_link(_opts \\ []) do
    :ets.new(@table, [:named_table, :public, :set, read_concurrency: true])
    :ignore
  end

  def child_spec(_opts) do
    %{id: __MODULE__, start: {__MODULE__, :start_link, []}, type: :worker, restart: :permanent}
  end

  def fetch_or_compute(key, fun, opts \\ []) do
    ttl_ms = Keyword.get(opts, :ttl_ms, 60_000)
    now = System.monotonic_time(:millisecond)

    case :ets.lookup(@table, key) do
      [{^key, value, expires_at}] when expires_at > now ->
        {:ok, value}

      _ ->
        case fun.() do
          {:ok, value} ->
            :ets.insert(@table, {key, value, now + ttl_ms})
            {:ok, value}

          other ->
            other
        end
    end
  end

  def clear, do: :ets.delete_all_objects(@table)
end

defmodule Verscienta.Gamification.Store do
  @moduledoc """
  Atomic spend + effect application for reward redemptions. All operations
  go through a single DB transaction so a partial failure rolls everything
  back.
  """
  require Ash.Query

  alias Verscienta.Accounts.User
  alias Verscienta.Engagement.Comment
  alias Verscienta.Gamification
  alias Verscienta.Gamification.GamificationAudit
  alias Verscienta.Gamification.Redemption
  alias Verscienta.Gamification.RewardItem
  alias Verscienta.Gamification.Streak
  alias Verscienta.Repo

  @rate_limit_per_hour 10

  def redeem(%User{} = user, %RewardItem{} = item, opts \\ []) do
    target_comment_id = Keyword.get(opts, :target_comment_id)

    Repo.transaction(fn ->
      with :ok <- check_rate_limit(user),
           {:ok, locked} <- lock_user(user.id),
           :ok <- check_funds(locked, item),
           :ok <- check_stock(user.id, item.id, item.stock_per_user),
           :ok <- check_cooldown(user.id, item.id, item.cooldown_hours),
           {:ok, redemption} <- insert_redemption(user.id, item),
           :ok <- write_spend(user.id, redemption.id, item.cost),
           :ok <- apply_effect(user.id, item, target_comment_id, redemption.id) do
        Gamification.broadcast(user.id, {:points_changed, user.id})
        Phoenix.PubSub.broadcast(Verscienta.PubSub, "user:#{user.id}", {:store_redeemed, user.id, redemption.id})
        redemption
      else
        {:error, reason} ->
          Repo.rollback(reason)
      end
    end)
  end

  defp check_rate_limit(user) do
    one_hour_ago = DateTime.add(DateTime.utc_now(), -3600, :second)

    count =
      Redemption
      |> Ash.Query.filter(user_id == ^user.id and redeemed_at >= ^one_hour_ago)
      |> Ash.read!(authorize?: false)
      |> length()

    if count >= @rate_limit_per_hour do
      write_audit(user.id, :store_rate_limit, :warn, %{"count" => count})
      {:error, :rate_limited}
    else
      :ok
    end
  end

  defp lock_user(user_id) do
    case Repo.query("SELECT id, points FROM users WHERE id = $1 FOR UPDATE", [Ecto.UUID.dump!(user_id)]) do
      {:ok, %Postgrex.Result{rows: [[_, points]]}} ->
        {:ok, %{id: user_id, points: points}}

      _ ->
        {:error, :not_found}
    end
  end

  defp check_funds(%{points: p}, %{cost: c}) when p >= c, do: :ok
  defp check_funds(_, _), do: {:error, :insufficient_points}

  defp check_stock(_, _, nil), do: :ok

  defp check_stock(user_id, item_id, limit) do
    count =
      Redemption
      |> Ash.Query.filter(user_id == ^user_id and reward_item_id == ^item_id)
      |> Ash.read!(authorize?: false)
      |> length()

    if count >= limit, do: {:error, :out_of_stock_for_user}, else: :ok
  end

  defp check_cooldown(_, _, nil), do: :ok

  defp check_cooldown(user_id, item_id, hours) do
    horizon = DateTime.add(DateTime.utc_now(), -hours * 3600, :second)

    recent =
      Redemption
      |> Ash.Query.filter(user_id == ^user_id and reward_item_id == ^item_id and redeemed_at >= ^horizon)
      |> Ash.read!(authorize?: false)
      |> length()

    if recent > 0 do
      write_audit(user_id, :store_cooldown_hit, :info, %{"item_id" => item_id})
      {:error, :on_cooldown}
    else
      :ok
    end
  end

  defp insert_redemption(user_id, item) do
    Redemption
    |> Ash.Changeset.for_create(:create, %{
      user_id: user_id,
      reward_item_id: item.id,
      cost_paid: item.cost,
      redeemed_at: DateTime.utc_now()
    })
    |> Ash.create(authorize?: false)
  end

  defp write_spend(user_id, redemption_id, cost) do
    :ok = Gamification.award(user_id, :redemption, "redemption:#{redemption_id}", -cost)
    :ok
  end

  defp apply_effect(user_id, %RewardItem{kind: :flair_color, key: key}, _, _) do
    unlock_flair(user_id, :unlocked_flair_color, key)
  end

  defp apply_effect(user_id, %RewardItem{kind: :flair_frame, key: key}, _, _) do
    unlock_flair(user_id, :unlocked_flair_frame, key)
  end

  defp apply_effect(user_id, %RewardItem{kind: :badge_cosmetic, key: key}, _, _) do
    unlock_flair(user_id, :unlocked_badge_cosmetic, key)
  end

  defp apply_effect(user_id, %RewardItem{kind: :theme, key: key}, _, _) do
    unlock_flair(user_id, :unlocked_theme, key)
  end

  defp apply_effect(user_id, %RewardItem{kind: :streak_freeze}, _, _) do
    streak =
      Streak
      |> Ash.Query.filter(user_id == ^user_id)

    otp_app: :verscienta,
    domain: Verscienta.Gamification,

  def statuses, do: @statuses

    table "referrals"
    repo Verscienta.Repo

    custom_indexes do
      index [:referrer_id, :status, :qualified_at]
    end
  end

  actions do
    defaults [:read]

    create :create do
      accept [:referrer_id, :referee_id, :code_used, :signup_ip, :status, :revoked_at, :revoked_reason]
    end

    update :qualify do
      accept []
      argument :now, :utc_datetime_usec, allow_nil?: false
      require_atomic? false
      change fn cs, _ ->
        cs
        |> Ash.Changeset.force_change_attribute(:status, :qualified)
        |> Ash.Changeset.force_change_attribute(:qualified_at, Ash.Changeset.get_argument(cs, :now))
      end

        |> Ash.Changeset.force_change_attribute(:revoked_at, Ash.Changeset.get_argument(cs, :now))
      end
    end
  end

    uuid_primary_key :id

defmodule Verscienta.Gamification.Referrals do
  @moduledoc "Code generation, signup capture, qualification, caps, and revocation."
  require Ash.Query

  alias Verscienta.Accounts.User
  alias Verscienta.Gamification
  alias Verscienta.Gamification.GamificationAudit
  alias Verscienta.Gamification.Referral

  @alphabet ~c"0123456789ABCDEFGHJKMNPQRSTVWXYZ"
  @code_length 8
  @monthly_cap 10

  def generate_code do
    1..@code_length
    |> Enum.map(fn _ -> Enum.random(@alphabet) end)
    |> List.to_string()
  end

  @doc "Resolve a referral code to a confirmed referrer user, or nil."
  def resolve(code) when is_binary(code) do
    User
    |> Ash.Query.filter(referral_code == ^code and not is_nil(confirmed_at))
    |> Ash.read_one(authorize?: false)
    |> case do
      {:ok, u} -> u
      _ -> nil
    end
  end

  def resolve(_), do: nil

  @doc """
  Capture a referral at signup. Returns `{:ok, %Referral{}}` with status
  reflecting any throttling / self-referral checks. Idempotent on (referrer_id, referee_id).
  """
  def capture(referee_id, code, signup_ip, opts \\ []) do
    case resolve(code) do
      nil -> {:error, :unknown_code}

      %User{id: rid} when rid == referee_id ->
        {:error, :self}

      referrer ->
        referrer_ip = Keyword.get(opts, :referrer_ip)

        attrs = %{
          referrer_id: referrer.id,
          referee_id: referee_id,
          code_used: code,
          signup_ip: signup_ip
        }

        cond do
          referrer_ip && referrer_ip == signup_ip ->
            do_create(attrs, :revoked, "self_referral_ip")

          ip_throttled?(signup_ip) ->
            do_create(attrs, :revoked, "ip_throttle")

          true ->
            with {:ok, r} <- do_create(attrs, :pending, nil) do
              assign_referrer(referee_id, referrer.id)
              {:ok, r}
            end
        end
    end
  end

    horizon = DateTime.add(DateTime.utc_now(), -86_400, :second)

    Referral
    |> Ash.Query.filter(signup_ip == ^ip and inserted_at >= ^horizon)
    |> Ash.read!(authorize?: false)
    |> length() >= 3
  end

  defp do_create(attrs, status, revoked_reason) do
    full_attrs =
      attrs
      |> Map.put(:status, status)
      |> Map.put(:revoked_at, if(status == :revoked, do: DateTime.utc_now(), else: nil))
      |> Map.put(:revoked_reason, revoked_reason)

    Referral
    |> Ash.Changeset.for_create(:create, full_attrs)
    |> Ash.create(authorize?: false)
  end

  defp assign_referrer(referee_id, referrer_id) do
    referee = Ash.get!(User, referee_id, authorize?: false)

    referee
    |> Ash.Changeset.for_update(:assign_referrer, %{referred_by_id: referrer_id})
    |> Ash.update!(authorize?: false)
  end

  @doc """
  Qualify a pending Referral for `referee_id` and award the referrer 100 pts
  unless the monthly cap is exceeded (in which case status flips but no
  points and an audit row is written).
  """
  def qualify_for_user(referee_id) do
    Referral
    |> Ash.Query.filter(referee_id == ^referee_id and status == :pending)
    |> Ash.read_one(authorize?: false)
    |> case do
      {:ok, nil} -> {:ok, :no_pending}

      {:ok, r} ->
        now = DateTime.utc_now()

        r
        |> Ash.Changeset.for_update(:qualify, %{now: now})
        |> Ash.update!(authorize?: false)

        if over_monthly_cap?(r.referrer_id, now) do
          write_audit(r.referrer_id, :referral_cap_hit, :info, %{"referral_id" => to_string(r.id)})
        else
          :ok = Gamification.award(r.referrer_id, :referral_bonus, "referral:#{r.id}", 100)
        end

        {:ok, r.id}

      err -> err
    end
  end

  defp over_monthly_cap?(referrer_id, now) do
    month_start = DateTime.new!(%Date{now |> DateTime.to_date() | day: 1}, ~T[00:00:00], "Etc/UTC")

    Referral
    |> Ash.Query.filter(
      referrer_id == ^referrer_id and status == :qualified and qualified_at >= ^month_start
    )
    |> Ash.read!(authorize?: false)
    |> length() > @monthly_cap
  end

  defp write_audit(user_id, kind, severity, details) do
    GamificationAudit
    |> Ash.Changeset.for_create(:create, %{
      user_id: user_id,
      kind: kind,
      severity: severity,
      details: details,
      occurred_at: DateTime.utc_now()
    })
    |> Ash.create!(authorize?: false)
  end

defmodule Verscienta.Gamification.Workers.AnomalyScanWorker do
  @moduledoc "Hourly cron: writes audit rows for users gaining >500 points in the last hour."
  use Oban.Worker, queue: :default, max_attempts: 3, unique: [period: 600]

  import Ecto.Query

  alias Verscienta.Gamification.GamificationAudit
  alias Verscienta.Repo

  @threshold 500

  @impl Oban.Worker
  def perform(%Oban.Job{}) do
    one_hour_ago = DateTime.add(DateTime.utc_now(), -3600, :second)

    query =
      from p in "points_entries",
        where: p.amount > 0 and p.inserted_at >= ^one_hour_ago,
        group_by: p.user_id,
        select: %{user_id: p.user_id, total: sum(p.amount)}

    Repo.all(query)
    |> Enum.filter(&(&1.total > @threshold))
    |> Enum.each(fn %{user_id: uid, total: total} ->
      GamificationAudit
      |> Ash.Changeset.for_create(:create, %{
        user_id: Ecto.UUID.cast!(uid),
        kind: :anomaly_point_spike,
        severity: :critical,
        details: %{"window_total" => total},
        occurred_at: DateTime.utc_now()
      })
      |> Ash.create(authorize?: false)
    end)

    :ok
  end
end

defmodule Verscienta.Gamification.Backfill do
  @moduledoc "One-shot backfill of GamificationEvent rows from existing PointsEntry history."
  require Ash.Query

  alias Verscienta.Accounts.User
  alias Verscienta.Gamification.GamificationEvent
  alias Verscienta.Gamification.PointsEntry

  def events_from_points_entries! do
    PointsEntry
    |> Ash.read!(authorize?: false)
    |> Enum.each(&insert_event/1)

    :ok
  end

  defp insert_event(entry) do
    user = Ash.get!(User, entry.user_id, authorize?: false)
    tz = user.timezone || "Etc/UTC"

    local_date =
      entry.inserted_at
      |> DateTime.shift_zone!(tz)
      |> DateTime.to_date()

    GamificationEvent
    |> Ash.Changeset.for_create(:create, %{
      user_id: entry.user_id,
      kind: entry.source,
      ref: entry.source_ref,
      metadata: %{},
      occurred_at: entry.inserted_at,
      local_date: local_date
    })
    |> Ash.create(authorize?: false)
  rescue
    _ -> :ok
  end
end

  end
end
