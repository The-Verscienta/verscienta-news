# Sponsored Content Placements Implementation Plan

> **Historical — standalone Verscienta app.** This file describes `Verscienta.*` LiveView work. It is not the Kiln overlay and it is not Slice 1. Current scope is [`docs/slice-1.md`](../../slice-1.md).


> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let admins/moderators mark an editorial `Article` as sponsored and surface a clear, FTC-compliant "Sponsored" disclosure on feed cards and above the article body in the reader.

**Architecture:** Three inline fields on the `Article` resource (`sponsored?`, `sponsor_name`, `sponsor_url`) set by a staff-only `:set_sponsorship` Ash action (policy + validation mirror the existing `:set_featured_until`). Disclosure renders via a new card badge in `news_card/1` and a new `LegalComponents.sponsored_disclosure/1` banner. Staff set/clear sponsorship from the reader page next to the existing feature/unfeature controls. Label only — no placement, ranking, or tracking changes.

**Tech Stack:** Elixir, Ash 3 + AshPostgres, Phoenix LiveView, ExUnit, Tailwind v4.

**Spec:** `docs/superpowers/specs/2026-06-15-sponsored-content-placements-design.md`

```elixir
defmodule Verscienta.Content.Article.Changes.NormalizeSponsorship do
  @moduledoc """
  Validates and normalizes the sponsorship fields on `:set_sponsorship`:

    * when the resulting `sponsored?` is true, `sponsor_name` must be present;
    * a present `sponsor_url` must be an http/https URL (rejects javascript:, data:, …);
    * when `sponsored?` is false, force sponsor_name and sponsor_url to nil.
  """
  use Ash.Resource.Change

  @impl true
  def change(changeset, _opts, _context) do
    sponsored? = Ash.Changeset.get_attribute(changeset, :sponsored?)

    if sponsored? do
      changeset
      |> validate_name()
      |> validate_url()
    else
      changeset
      |> Ash.Changeset.force_change_attribute(:sponsor_name, nil)
      |> Ash.Changeset.force_change_attribute(:sponsor_url, nil)
    end
  end

  defp validate_name(changeset) do
    case changeset |> Ash.Changeset.get_attribute(:sponsor_name) |> blank?() do
      true ->
        Ash.Changeset.add_error(changeset,
          field: :sponsor_name,
          message: "is required when an article is sponsored"
        )

      false ->
        changeset
    end
  end

  defp validate_url(changeset) do
    case Ash.Changeset.get_attribute(changeset, :sponsor_url) do
      nil ->
        changeset

      "" ->
        Ash.Changeset.force_change_attribute(changeset, :sponsor_url, nil)

      url ->
        if http_url?(url) do
          changeset
        else
          Ash.Changeset.add_error(changeset,
            field: :sponsor_url,
            message: "must be an http or https URL"
          )
        end
    end
  end

  defp http_url?(url) do
    case URI.parse(to_string(url)) do
      %URI{scheme: scheme, host: host}
      when is_binary(scheme) and is_binary(host) and host != "" ->
        String.downcase(scheme) in ["http", "https"]

      _ ->
        false
    end
  end

  defp blank?(nil), do: true
  defp blank?(value), do: String.trim(to_string(value)) == ""
end
```

  defp set_sponsorship(socket, attrs) do
    actor = socket.assigns.current_user

    case socket.assigns.article
         |> Ash.Changeset.for_update(:set_sponsorship, attrs, actor: actor)
         |> Ash.update(actor: actor) do
      {:ok, article} ->
        article =
          Ash.load!(article, [:author, :tags, :source, :featured_now],
            actor: actor,
            authorize?: false
          )

        article = if socket.assigns.premium_gated?, do: %{article | blocks: []}, else: article

        {:noreply,
         socket
         |> assign(article: article)
         |> put_flash(:info, "Updated")}

      {:error, _} ->
        {:noreply, put_flash(socket, :error, "Could not update sponsorship")}
    end
  end

  defp blank_to_nil(nil), do: nil
  defp blank_to_nil(value), do: if(String.trim(value) == "", do: nil, else: value)
